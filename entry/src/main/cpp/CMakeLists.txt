cmake_minimum_required(VERSION 3.16)
project(tinybot_sandbox LANGUAGES C CXX)
option(SANDBOX_DEVICE_TESTS "Build the native test executable for a connected HarmonyOS device" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
if(DEFINED PACKAGE_FIND_FILE)
    include(${PACKAGE_FIND_FILE})
endif()
set(QJS third_party/quickjs)
add_library(tinybot_quickjs STATIC
    ${QJS}/quickjs.c ${QJS}/dtoa.c ${QJS}/libregexp.c ${QJS}/libunicode.c ${QJS}/cutils.c)
set_target_properties(tinybot_quickjs PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(tinybot_quickjs PUBLIC ${QJS})
target_compile_definitions(tinybot_quickjs PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
target_compile_options(tinybot_quickjs PRIVATE -fwrapv)
add_library(sandbox_core STATIC sandbox.cpp)
set_target_properties(sandbox_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(sandbox_core PUBLIC tinybot_quickjs)
target_include_directories(sandbox_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    add_library(tinybot_sandbox SHARED napi_sandbox.cpp napi_orchestrator.cpp)
    target_link_libraries(tinybot_sandbox PRIVATE sandbox_core libace_napi.z.so)
    set(linux_assets "${CMAKE_CURRENT_SOURCE_DIR}/../resources/rawfile/linux")
    file(GLOB linux_engine CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/../../../libs/${OHOS_ARCH}/libqemu-system-aarch64-tci.so")
    if(linux_engine)
        if(NOT EXISTS "${linux_assets}/kernel" OR NOT EXISTS "${linux_assets}/root-v5-shell.qcow2")
            message(FATAL_ERROR "Prepare the Linux guest with scripts/linux-runtime/build-guest.sh")
        endif()
        target_sources(tinybot_sandbox PRIVATE napi_linux.cpp napi_linux_files.cpp)
        target_compile_definitions(tinybot_sandbox PRIVATE TINYBOT_LINUX_RUNTIME=1)
        target_link_libraries(tinybot_sandbox PRIVATE libchild_process.so)
        add_library(tinybot_linux_child SHARED linux_guest.cpp)
    endif()
else()
    find_package(Threads REQUIRED)
    target_link_libraries(sandbox_core PUBLIC Threads::Threads)
    if(NOT WIN32)
        target_link_libraries(sandbox_core PUBLIC m)
    endif()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS" OR SANDBOX_DEVICE_TESTS)
    enable_testing()
    add_executable(native_job_result_tests tests/native_job_result_test.cpp)
    add_test(NAME native_job_result COMMAND native_job_result_tests)
    add_executable(sandbox_tests tests/sandbox_test.cpp)
    target_link_libraries(sandbox_tests PRIVATE sandbox_core)
    add_test(NAME sandbox COMMAND sandbox_tests)
    add_executable(orchestration_tests tests/orchestration_test.cpp)
    target_link_libraries(orchestration_tests PRIVATE sandbox_core)
    add_test(NAME orchestration COMMAND orchestration_tests)
    set_tests_properties(sandbox orchestration native_job_result PROPERTIES TIMEOUT 45)
endif()

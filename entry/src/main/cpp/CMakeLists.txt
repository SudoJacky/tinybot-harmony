cmake_minimum_required(VERSION 3.16)
project(tinybot_sandbox LANGUAGES C CXX)
option(SANDBOX_DEVICE_TESTS "Build the native test executable for a connected HarmonyOS device" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
if(DEFINED PACKAGE_FIND_FILE)
    include(${PACKAGE_FIND_FILE})
endif()
set(QJS third_party/quickjs)
add_library(tinybot_quickjs STATIC
    ${QJS}/quickjs.c ${QJS}/dtoa.c ${QJS}/libregexp.c ${QJS}/libunicode.c ${QJS}/cutils.c)
set_target_properties(tinybot_quickjs PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(tinybot_quickjs PUBLIC ${QJS})
target_compile_definitions(tinybot_quickjs PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
target_compile_options(tinybot_quickjs PRIVATE -fwrapv)
add_library(sandbox_core STATIC sandbox.cpp)
set_target_properties(sandbox_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(sandbox_core PUBLIC tinybot_quickjs)
target_include_directories(sandbox_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    add_library(tinybot_sandbox SHARED napi_sandbox.cpp)
    target_link_libraries(tinybot_sandbox PRIVATE sandbox_core libace_napi.z.so)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(sandbox_core PUBLIC Threads::Threads)
    if(NOT WIN32)
        target_link_libraries(sandbox_core PUBLIC m)
    endif()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS" OR SANDBOX_DEVICE_TESTS)
    enable_testing()
    add_executable(sandbox_tests tests/sandbox_test.cpp)
    target_link_libraries(sandbox_tests PRIVATE sandbox_core)
    add_test(NAME sandbox COMMAND sandbox_tests)
    set_tests_properties(sandbox PROPERTIES TIMEOUT 45)
endif()
